14 · Evaluation, Metrics & Persistence

Evaluator eval = new Evaluator(new String[]{"cat","dog","bird"});
Evaluator auto = new Evaluator(); // infers num classes

EvaluationResult res = eval.evaluate(net, testImages, testLabels);
System.out.println(res); // accuracy, correct/total
System.out.println(res.accuracy()); // 0.0–1.0
ConfusionMatrix cm = res.confusionMatrix();
cm.print(); // ASCII table
cm.printClassMetrics(classNames); // precision/recall/F1 per class
int correct = cm.correct();
int[][] matrix = cm.toArray();

// Fine-grained predictions
PredictionResult pr = eval.predict(net, testImages, testLabels);
for(Prediction p: pr.getPredictions()){
  int pred = p.predictedLabel();
  int truth= p.actualLabel();
  double conf= p.confidence();
}
Prediction single = pr.get(0);
