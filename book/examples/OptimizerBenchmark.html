OptimizerBenchmark

All 6 optimizers head-to-head - Synthetic 5-class Gaussian mixture (20-D). Trains identical 20-64-32-5 + Softmax nets with SGD, SGD+mom, Adam, AdamW, RMSProp, AdaGrad for 20 epochs and prints accuracy/time.

ML principles

  • Controlled optimizer comparison: same synthetic 5-class mixture (Gaussian + label shift), same net 20-64-32-5, same 20 epochs.
  • All 6 variants: SGD, SGD+mom0.9, Adam, AdamW, RMSProp, AdaGrad.
X = Tensor(2000,20) fill N(0,1)+label*0.5; y=rand 0..4
split 80/20
for opt in [SGD(0.01), SGD(0.01,0.9), Adam(0.001), AdamW(0.001,0.01), RMSProp(0.001), AdaGrad(0.01)]:
  build fresh net 20-64 ReLU -64-32 ReLU -32-5 Softmax
  trainer.fit(trainX,trainY,20,64); evaluate test acc
Expected: Adam/AdamW fastest to converge; AdaGrad slows after many steps (cache grows).

Full source - src/main/java/examples/OptimizerBenchmark.java

package examples;

import ml.core.Tensor;
import ml.core.SeedManager;
import ml.network.NeuralNetwork;
import ml.training.*;
import ml.optimizer.*;
import ml.loss.*;
import ml.evaluation.*;
import ml.regularization.*;
import ml.activation.*;
import java.util.Random;

public class OptimizerBenchmark {
    public static void main(String[] args) {
        SeedManager.setSeed(42);
        System.out.println("=== Optimizer Benchmark ===\n");

        int samples = 2000;
        int features = 20;
        int classes = 5;
        Random rng = new Random(42);

        Tensor X = new Tensor(samples, features);
        int[] y = new int[samples];
        for (int i = 0; i < samples; i++) {
            int label = rng.nextInt(classes);
            y[i] = label;
            for (int j = 0; j < features; j++)
                X.data()[i * features + j] = rng.nextGaussian() + label * 0.5;
        }

        int split = (int) (samples * 0.8);
        Tensor trainX = X.sliceRows(0, split);
        int[] trainY = java.util.Arrays.copyOf(y, split);
        Tensor testX = X.sliceRows(split, samples);
        int[] testY = java.util.Arrays.copyOfRange(y, split, samples);

        Optimizer[] optimizers = {
            new SGD(0.01),
            new SGD(0.01, 0.9),
            new Adam(0.001),
            new AdamW(0.001, 0.01),
            new RMSProp(0.001),
            new AdaGrad(0.01)
        };

        for (Optimizer opt : optimizers) {
            System.out.println("--- " + opt.getClass().getSimpleName() + " ---");
            NeuralNetwork net = new NeuralNetwork();
            net.add(new ml.layer.DenseLayer(features, 64));
            net.add(new ReLU());
            net.add(new ml.layer.DenseLayer(64, 32));
            net.add(new ReLU());
            net.add(new ml.layer.DenseLayer(32, classes));
            net.add(new Softmax());

            CrossEntropyLoss loss = new CrossEntropyLoss();
            Trainer trainer = new Trainer(net, loss, opt);
            TrainingHistory history = trainer.fit(trainX, trainY, 20, 64);

            Evaluator evaluator = new Evaluator();
            EvaluationResult result = evaluator.evaluate(net, testX, testY);
            System.out.printf("  Final loss: %.6f | Accuracy: %.2f%% | Time: %dms%n%n",
                history.lastLoss(), result.accuracyPercentage(), history.totalTimeMs());
        }
    }
}
