// P01 MnistClassifier — MLP on 28×28 (784 → 256 → 128 → 64 → 10)
NeuralNetwork mlp = new NeuralNetwork()
  .add(new DenseLayer(784,256)).add(new ReLU()).add(new Dropout(0.2))
  .add(new DenseLayer(256,128)).add(new ReLU()).add(new Dropout(0.2))
  .add(new DenseLayer(128,64)).add(new ReLU())
  .add(new DenseLayer(64,10)).add(new Softmax());
new Trainer(mlp, new CrossEntropyLoss(), new Adam(0.001))
  .withGradientClipper(new GradientClipper(5.0)).fit(trainX,trainY,10,256);

// P01b FashionMnistClassifier — CNN: Conv→Pool→Conv→Pool→Dense
NeuralNetwork cnn = new NeuralNetwork()
  .add(new Conv2D(1,16,3)).add(new ReLU()).add(new MaxPool(2))
  .add(new Conv2D(16,32,3)).add(new ReLU()).add(new MaxPool(2))
  .add(new Flatten()).add(new DenseLayer(1568,128)).add(new ReLU())
  .add(new DenseLayer(128,10)).add(new Softmax());

// P04 MnistAutoencoder — Denoising AE 784→256→64→256→784 with MSELoss
NeuralNetwork ae = new NeuralNetwork()
  .add(new DenseLayer(784,256)).add(new ReLU())
  .add(new DenseLayer(256,64)).add(new ReLU())   // bottleneck
  .add(new DenseLayer(64,256)).add(new ReLU())
  .add(new DenseLayer(256,784)).add(new Sigmoid());
new Trainer(ae, new MSELoss(), new Adam(0.002)).fit(noisyX, cleanY, 10, 256);
