Quickstart - 30 seconds

NeuralNetwork net = new NeuralNetwork()
  .add(new DenseLayer(784, 256))
  .add(new ReLU())
  .add(new Dropout(0.2))
  .add(new DenseLayer(256, 10))
  .add(new Softmax());
Trainer trainer = new Trainer(net, new CrossEntropyLoss(), new Adam(0.001));
trainer.fit(trainImages, trainLabels, 10, 256);
System.out.println(new Evaluator(classNames).evaluate(net, testImages, testLabels));

Build: mvn compile exec:java -Dexec.mainClass=examples.MNISTClassification
