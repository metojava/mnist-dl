package ml;

import java.awt.BasicStroke;
import java.awt.Color;
import java.awt.Font;
import java.awt.Graphics2D;
import java.awt.RenderingHints;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.IOException;

import javax.imageio.ImageIO;

/**
 * Visualization utilities for Fashion-MNIST predictions.
 *
 * Fashion-MNIST labels:
 *
 * 0 = T-shirt/top
 * 1 = Trouser
 * 2 = Pullover
 * 3 = Dress
 * 4 = Coat
 * 5 = Sandal
 * 6 = Shirt
 * 7 = Sneaker
 * 8 = Bag
 * 9 = Ankle boot
 *
 * This class creates PNG images showing:
 *
 *   - Fashion-MNIST image
 *   - Predicted class
 *   - Prediction confidence
 *   - Actual class
 *   - Correct / incorrect status
 */
public class MNISTVisualizer {

    /**
     * Fashion-MNIST class names.
     */
    public static final String[] FASHION_LABELS = {
            "T-shirt/top",
            "Trouser",
            "Pullover",
            "Dress",
            "Coat",
            "Sandal",
            "Shirt",
            "Sneaker",
            "Bag",
            "Ankle boot"
    };

    /**
     * Creates a PNG containing multiple Fashion-MNIST
     * predictions arranged in a grid.
     *
     * @param network trained neural network
     * @param images image tensor
     * @param labels actual labels as int[]
     * @param startIndex first image to display
     * @param count number of images to display
     * @param outputFile output PNG filename
     */
    public static void savePredictions(
            NeuralNetwork network,
            Tensor images,
            int[] labels,
            int startIndex,
            int count,
            String outputFile
    ) throws IOException {

        if (network == null) {
            throw new IllegalArgumentException(
                    "network cannot be null"
            );
        }

        if (images == null) {
            throw new IllegalArgumentException(
                    "images cannot be null"
            );
        }

        if (labels == null) {
            throw new IllegalArgumentException(
                    "labels cannot be null"
            );
        }

        /*
         * Fashion-MNIST images are:
         *
         * 28 x 28 = 784 pixels
         */
        if (images.cols() != 784) {
            throw new IllegalArgumentException(
                    "Expected Fashion-MNIST images with 784 " +
                    "features, but found " +
                    images.cols()
            );
        }

        if (labels.length != images.rows()) {
            throw new IllegalArgumentException(
                    "Number of labels (" +
                    labels.length +
                    ") does not match number of images (" +
                    images.rows() +
                    ")"
            );
        }

        if (startIndex < 0) {
            throw new IllegalArgumentException(
                    "startIndex cannot be negative"
            );
        }

        if (startIndex >= images.rows()) {
            throw new IllegalArgumentException(
                    "startIndex " +
                    startIndex +
                    " is outside the dataset"
            );
        }

        if (count <= 0) {
            throw new IllegalArgumentException(
                    "count must be greater than zero"
            );
        }

        /*
         * Don't go past the end of the dataset.
         */
        int actualCount =
                Math.min(
                        count,
                        images.rows() - startIndex
                );

        /*
         * Grid configuration.
         */
        int columns = 4;

        int rows =
                (int) Math.ceil(
                        actualCount /
                        (double) columns
                );

        /*
         * Size of each prediction card.
         */
        int cellWidth = 360;
        int cellHeight = 390;

        int totalWidth =
                columns * cellWidth;

        int totalHeight =
                rows * cellHeight;

        BufferedImage output =
                new BufferedImage(
                        totalWidth,
                        totalHeight,
                        BufferedImage.TYPE_INT_RGB
                );

        Graphics2D graphics =
                output.createGraphics();

        /*
         * Enable high-quality text rendering.
         */
        graphics.setRenderingHint(
                RenderingHints.KEY_ANTIALIASING,
                RenderingHints.VALUE_ANTIALIAS_ON
        );

        graphics.setRenderingHint(
                RenderingHints.KEY_TEXT_ANTIALIASING,
                RenderingHints.VALUE_TEXT_ANTIALIAS_ON
        );

        /*
         * White background.
         */
        graphics.setColor(Color.WHITE);

        graphics.fillRect(
                0,
                0,
                totalWidth,
                totalHeight
        );

        /*
         * Draw every prediction.
         */
        for (int i = 0;
             i < actualCount;
             i++) {

            int imageIndex =
                    startIndex + i;

            int column =
                    i % columns;

            int row =
                    i / columns;

            int x =
                    column * cellWidth;

            int y =
                    row * cellHeight;

            drawPrediction(
                    graphics,
                    network,
                    images,
                    labels,
                    imageIndex,
                    x,
                    y,
                    cellWidth,
                    cellHeight
            );
        }

        graphics.dispose();

        /*
         * Create parent directories if necessary.
         */
        File file =
                new File(outputFile);

        File parent =
                file.getParentFile();

        if (parent != null &&
            !parent.exists()) {

            if (!parent.mkdirs() &&
                !parent.exists()) {

                throw new IOException(
                        "Could not create directory: " +
                        parent
                );
            }
        }

        /*
         * Write PNG.
         */
        boolean written =
                ImageIO.write(
                        output,
                        "png",
                        file
                );

        if (!written) {
            throw new IOException(
                    "Could not write PNG file: " +
                    outputFile
            );
        }
    }

    /**
     * Draws one prediction card.
     */
    private static void drawPrediction(
            Graphics2D graphics,
            NeuralNetwork network,
            Tensor images,
            int[] labels,
            int imageIndex,
            int x,
            int y,
            int width,
            int height
    ) {

        /*
         * ---------------------------------------------------------
         * Card background
         * ---------------------------------------------------------
         */

        graphics.setColor(
                new Color(
                        248,
                        248,
                        248
                )
        );

        graphics.fillRect(
                x,
                y,
                width,
                height
        );

        /*
         * Card border.
         */
        graphics.setColor(
                new Color(
                        210,
                        210,
                        210
                )
        );

        graphics.setStroke(
                new BasicStroke(1.0f)
        );

        graphics.drawRect(
                x,
                y,
                width - 1,
                height - 1
        );

        /*
         * ---------------------------------------------------------
         * Get image pixels
         * ---------------------------------------------------------
         */

        double[] imageData =
                extractImage(
                        images,
                        imageIndex
                );

        /*
         * ---------------------------------------------------------
         * Create input tensor
         * ---------------------------------------------------------
         */

        Tensor input =
                new Tensor(
                        1,
                        784
                );

        System.arraycopy(
                imageData,
                0,
                input.data(),
                0,
                784
        );

        /*
         * ---------------------------------------------------------
         * Run network
         * ---------------------------------------------------------
         */

        Tensor probabilities =
                network.forward(input);

        double[] probabilitiesData =
                probabilities.data();

        /*
         * ---------------------------------------------------------
         * Find predicted class
         * ---------------------------------------------------------
         */

        int predictedClass =
                argmax(
                        probabilitiesData
                );

        double confidence =
                probabilitiesData[
                        predictedClass
                ];

        /*
         * ---------------------------------------------------------
         * Actual Fashion-MNIST class
         * ---------------------------------------------------------
         *
         * IMPORTANT:
         *
         * labels is int[], not Tensor.
         */
        int actualClass =
                labels[imageIndex];

        /*
         * Validate labels.
         */
        if (actualClass < 0 ||
            actualClass >= FASHION_LABELS.length) {

            actualClass = -1;
        }

        /*
         * ---------------------------------------------------------
         * Correct?
         * ---------------------------------------------------------
         */

        boolean correct =
                predictedClass == actualClass;

        /*
         * ---------------------------------------------------------
         * Draw image
         * ---------------------------------------------------------
         */

        int imageX =
                x + 50;

        int imageY =
                y + 20;

        int imageSize =
                260;

        drawFashionImage(
                graphics,
                imageData,
                imageX,
                imageY,
                imageSize
        );

        /*
         * ---------------------------------------------------------
         * Draw prediction information
         * ---------------------------------------------------------
         */

        int textX =
                x + 20;

        int textY =
                y + 305;

        /*
         * Prediction color.
         *
         * Green = correct
         * Red   = incorrect
         */
        if (correct) {

            graphics.setColor(
                    new Color(
                            0,
                            130,
                            60
                    )
            );

        } else {

            graphics.setColor(
                    new Color(
                            200,
                            40,
                            40
                    )
            );
        }

        graphics.setFont(
                new Font(
                        "Arial",
                        Font.BOLD,
                        16
                )
        );

        graphics.drawString(
                "Predicted: " +
                labelName(predictedClass),
                textX,
                textY
        );

        /*
         * Confidence.
         */
        graphics.setColor(
                Color.DARK_GRAY
        );

        graphics.setFont(
                new Font(
                        "Arial",
                        Font.PLAIN,
                        15
                )
        );

        graphics.drawString(
                String.format(
                        "Confidence: %.2f%%",
                        confidence * 100.0
                ),
                textX,
                textY + 25
        );

        /*
         * Actual label.
         */
        graphics.drawString(
                "Actual: " +
                labelName(actualClass),
                textX,
                textY + 50
        );

        /*
         * Correct / incorrect.
         */
        graphics.setFont(
                new Font(
                        "Arial",
                        Font.BOLD,
                        14
                )
        );

        if (correct) {

            graphics.setColor(
                    new Color(
                            0,
                            130,
                            60
                    )
            );

            graphics.drawString(
                    "✓ Correct",
                    x + width - 100,
                    y + 25
            );

        } else {

            graphics.setColor(
                    new Color(
                            200,
                            40,
                            40
                    )
            );

            graphics.drawString(
                    "✗ Incorrect",
                    x + width - 110,
                    y + 25
            );
        }
    }

    /**
     * Draw a 28x28 Fashion-MNIST image.
     *
     * The source image is assumed to contain normalized
     * pixel values in the range [0, 1].
     *
     * Values in [0, 255] are also supported.
     */
    private static void drawFashionImage(
            Graphics2D graphics,
            double[] pixels,
            int x,
            int y,
            int size
    ) {

        BufferedImage image =
                new BufferedImage(
                        28,
                        28,
                        BufferedImage.TYPE_BYTE_GRAY
                );

        for (int row = 0;
             row < 28;
             row++) {

            for (int col = 0;
                 col < 28;
                 col++) {

                int index =
                        row * 28 + col;

                double value =
                        pixels[index];

                /*
                 * If normalized, convert:
                 *
                 * 0.0 -> 0
                 * 1.0 -> 255
                 */
                if (value >= 0.0 &&
                    value <= 1.0) {

                    value *= 255.0;
                }

                value =
                        Math.max(
                                0.0,
                                Math.min(
                                        255.0,
                                        value
                                )
                        );

                int gray =
                        (int) Math.round(value);

                int rgb =
                        (gray << 16)
                        | (gray << 8)
                        | gray;

                image.setRGB(
                        col,
                        row,
                        rgb
                );
            }
        }

        /*
         * Nearest-neighbor scaling keeps the individual
         * Fashion-MNIST pixels sharp.
         */
        Object oldInterpolation =
                graphics.getRenderingHint(
                        RenderingHints.KEY_INTERPOLATION
                );

        graphics.setRenderingHint(
                RenderingHints.KEY_INTERPOLATION,
                RenderingHints.VALUE_INTERPOLATION_NEAREST_NEIGHBOR
        );

        graphics.drawImage(
                image,
                x,
                y,
                size,
                size,
                null
        );

        /*
         * Restore interpolation setting.
         */
        if (oldInterpolation != null) {

            graphics.setRenderingHint(
                    RenderingHints.KEY_INTERPOLATION,
                    oldInterpolation
            );
        }

        /*
         * Image border.
         */
        graphics.setColor(
                new Color(
                        100,
                        100,
                        100
                )
        );

        graphics.drawRect(
                x,
                y,
                size,
                size
        );
    }

    /**
     * Extract one 784-value image from the image tensor.
     */
    private static double[] extractImage(
            Tensor images,
            int imageIndex
    ) {

        double[] source =
                images.data();

        double[] image =
                new double[784];

        int offset =
                imageIndex * 784;

        System.arraycopy(
                source,
                offset,
                image,
                0,
                784
        );

        return image;
    }

    /**
     * Return index of largest value.
     */
    private static int argmax(
            double[] values
    ) {

        if (values == null ||
            values.length == 0) {

            throw new IllegalArgumentException(
                    "Cannot calculate argmax of empty array"
            );
        }

        int bestIndex =
                0;

        double bestValue =
                values[0];

        for (int i = 1;
             i < values.length;
             i++) {

            if (values[i] > bestValue) {

                bestValue =
                        values[i];

                bestIndex =
                        i;
            }
        }

        return bestIndex;
    }

    /**
     * Convert numeric Fashion-MNIST class to
     * human-readable class name.
     */
    public static String labelName(
            int label
    ) {

        if (label < 0 ||
            label >= FASHION_LABELS.length) {

            return "Unknown (" +
                    label +
                    ")";
        }

        return FASHION_LABELS[label];
    }

    /**
     * Return a copy of the Fashion-MNIST labels.
     */
    public static String[] getFashionLabels() {

        return FASHION_LABELS.clone();
    }
}

Important change

The key API now matches your actual loader:

public static void savePredictions(
        NeuralNetwork network,
        Tensor images,
        int[] labels,
        int startIndex,
        int count,
        String outputFile
)


So this existing code in MNISTExample.java should compile:

MNISTVisualizer.savePredictions(
        network,
        test.images(),
        test.labels(),
        0,
        numberOfPredictions,
        predictionsImage
);


There is no Tensor conversion for labels.

Your data flow is now:

Fashion-MNIST
      │
      ├── images → Tensor [N × 784]
      │
      └── labels → int[] [N]
                       │
                       ▼
              MNISTVisualizer
                       │
             ┌─────────┴─────────┐
             ▼                   ▼
        numeric label        class name
             7                   │
             │                   ▼
             └─────────────► "Sneaker"


And predictions.png will show names such as Sneaker, Ankle boot, Trouser, Bag, etc., rather than numeric digit labels.

One additional point: this version has no JFreeChart dependency/imports, because prediction-image generation doesn't need JFreeChart. Your TrainingChart.java can continue using JFreeChart separately.
